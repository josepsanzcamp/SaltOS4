
# Use composer to get the library:

rm -rf vendor
rm -f composer.*
echo '{}' > composer.json
composer config platform.php 8.2.0
composer require phpoffice/phpspreadsheet
composer outdated --direct --ignore-platform-reqs

composer show phpoffice/phpspreadsheet --format=json | jq -r '.versions[0]' | sed 's/^v//' > VERSION

# To fix the permissions in files and directories, use it:

find vendor -type d | gawk '{print "\""$0"\""}' | xargs chmod 0775
find vendor -type f | gawk '{print "\""$0"\""}' | xargs chmod 0664
