version=$(curl -s https://data.jsdelivr.com/v1/packages/gh/chartjs/Chart.js/resolved | jq -r .version)
echo $version > VERSION

rm -f chart.umd.js
wget https://cdn.jsdelivr.net/npm/chart.js@$version/dist/chart.umd.js

cat chart.umd.js | grep -v sourceMappingURL > chart.umd.min.js
rm -f chart.umd.js
