
# https://cdnjs.com/libraries/codemirror

version=$(curl -s https://api.cdnjs.com/libraries/codemirror?fields=version | jq -r .version)
echo $version > VERSION

wget -O js/codemirror.min.js        https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/codemirror.min.js
wget -O css/codemirror.min.css      https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/codemirror.min.css

wget -O js/mode/css.min.js          https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/mode/css/css.min.js
wget -O js/mode/javascript.min.js   https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/mode/javascript/javascript.min.js
wget -O js/mode/php.min.js          https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/mode/php/php.min.js
wget -O js/mode/sql.min.js          https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/mode/sql/sql.min.js
wget -O js/mode/xml.min.js          https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/mode/xml/xml.min.js
wget -O js/mode/yaml.min.js         https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/mode/yaml/yaml.min.js

wget -O js/addon/active-line.min.js https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/addon/selection/active-line.min.js

wget -O css/theme/dracula.min.css   https://cdnjs.cloudflare.com/ajax/libs/codemirror/$version/theme/dracula.min.css
