
# https://veno.es/venobox/
# https://cdnjs.com/libraries/venobox

version=$(curl -s https://api.cdnjs.com/libraries/venobox?fields=version | jq -r .version)
echo $version > VERSION

rm -f venobox.min.js
wget https://cdnjs.cloudflare.com/ajax/libs/venobox/$version/venobox.min.js

rm -f venobox.min.css
wget https://cdnjs.cloudflare.com/ajax/libs/venobox/$version/venobox.min.css

for i in venobox.min.js venobox.min.css; do
    cat $i | grep -v sourceMappingURL > $i.new
    mv $i.new $i
done
