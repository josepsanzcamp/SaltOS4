
version=$(curl -s https://data.jsdelivr.com/v1/packages/npm/bootstrap/resolved | jq -r .version)
echo $version > VERSION

rm -f bootstrap.bundle.min.js
wget https://cdn.jsdelivr.net/npm/bootstrap@$version/dist/js/bootstrap.bundle.min.js

rm -f bootstrap.min.css
wget https://cdn.jsdelivr.net/npm/bootstrap@$version/dist/css/bootstrap.min.css

for i in bootstrap.bundle.min.js bootstrap.min.css; do
    cat $i | grep -v sourceMappingURL > $i.new
    mv $i.new $i
done
