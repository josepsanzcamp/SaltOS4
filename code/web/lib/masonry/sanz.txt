# https://masonry.desandro.com/
# https://cdnjs.com/libraries/masonry

version=$(curl -s https://api.cdnjs.com/libraries/masonry?fields=version | jq -r .version)
echo $version > VERSION

rm -f masonry.pkgd.min.js
wget https://cdnjs.cloudflare.com/ajax/libs/masonry/$version/masonry.pkgd.min.js
