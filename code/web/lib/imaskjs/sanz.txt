
version=$(curl -s https://api.cdnjs.com/libraries/imask?fields=version | jq -r .version)
echo $version > VERSION

rm -f imask.min.js
wget https://cdnjs.cloudflare.com/ajax/libs/imask/$version/imask.min.js

cat imask.min.js | grep -v sourceMappingURL > imask.min.js.new
mv imask.min.js.new imask.min.js
